import java.io.IOException;
import java.io.PrintWriter;

import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

@WebServlet("/webapp")
public class WebApp extends HttpServlet {

    protected void doPost(HttpServletRequest request,
                           HttpServletResponse response)
            throws ServletException, IOException {

        PrintWriter out = response.getWriter();

        String uname = request.getParameter("username");
        String rno = request.getParameter("regno");
        String bran = request.getParameter("branch");
        String bat = request.getParameter("batch");
        String yy = request.getParameter("year");

        if (uname.isEmpty() || rno.isEmpty() || bran.isEmpty()
                || bat.isEmpty() || yy.isEmpty()) {

            out.println("<html><body>");
            out.println("<b>FILL ALL THE FIELDS</b><br>");
            out.println("</body></html>");

        } else {

            out.println("<html><body>");
            out.println("<b>Registered Successfully</b><br>");
            out.println("Name: " + uname + "<br>");
            out.println("Register Number: " + rno + "<br>");
            out.println("Branch: " + bran + "<br>");
            out.println("Batch: " + bat + "<br>");
            out.println("Year: " + yy + "<br>");
            out.println("</body></html>");
        }
    }
}
